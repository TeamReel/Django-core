from src.activities.models import Period
from src.organisations.models import Project

# Check Ajax's teams
ajax = Project.objects.filter(name='Ajax').first()
if ajax:
    print(f"\\nAjax ID: {ajax.id}")
    ajax_teams = Project.objects.filter(parent_project=ajax)
    print(f"Ajax has {ajax_teams.count()} child teams")

    for team in ajax_teams[:10]:
        period_count = Period.objects.filter(project=team).count()
        print(f"  - {team.name} (ID {team.id}): {period_count} periods")

# Check Almere City's teams
almere = Project.objects.filter(name='Almere City').first()
if almere:
    print(f"\\nAlmere City ID: {almere.id}")
    almere_teams = Project.objects.filter(parent_project=almere)
    print(f"Almere City has {almere_teams.count()} child teams")

    for team in almere_teams[:10]:
        period_count = Period.objects.filter(project=team).count()
        print(f"  - {team.name} (ID {team.id}): {period_count} periods")

# Total periods stats
total_periods = Period.objects.count()
periods_with_project = Period.objects.filter(project__isnull=False).count()
print(f"\\nTotal periods in DB: {total_periods}")
print(f"Periods with project assigned: {periods_with_project}")
